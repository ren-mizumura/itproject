-- 1. データベースの作成
CREATE DATABASE IF NOT EXISTS bbs_db CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;

USE bbs_db;

-- 2. テーブルの作成
CREATE TABLE IF NOT EXISTS posts (
id INT AUTO_INCREMENT PRIMARY KEY,
nickname VARCHAR(50) NOT NULL,
message TEXT NOT NULL,
media_path VARCHAR(255) DEFAULT NULL, -- ファイル保存パス
media_type VARCHAR(50) DEFAULT NULL,  -- 'image' または 'video'
created_at DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;
